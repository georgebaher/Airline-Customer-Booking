//1. Query 1 - Top 5 airport-to-airport routes by flight count
MATCH (f:Flight)-[:DEPARTS_FROM]->(o:Airport),
      (f)-[:ARRIVES_AT]->(d:Airport)
WITH o.station_code AS origin, d.station_code AS destination, count(f) AS flight_count
RETURN origin, destination, flight_count
ORDER BY flight_count DESC
LIMIT 5;

//2. Query 2 - Top 10 flights with the most passenger feedback
MATCH (:Passenger)-[:TOOK]->(j:Journey)-[:ON]->(f:Flight)
WITH f.flight_number + '/' + f.fleet_type_description AS flight_id,
     count(j) AS feedback_count
RETURN flight_id, feedback_count
ORDER BY feedback_count DESC
LIMIT 10;

//3. Query 3 - Average food satisfaction for multi-leg journeys grouped by generation
MATCH (p:Passenger)-[:TOOK]->(j:Journey)
WHERE j.number_of_legs > 1
WITH p.generation AS generation,
     count(j) AS multi_leg_count,
     avg(j.food_satisfaction_score) AS avg_score
RETURN generation, multi_leg_count, avg_score
ORDER BY multi_leg_count DESC;

//4. Query 4 - Flights with the shortest average arrival delays (top 10)
MATCH (j:Journey)-[:ON]->(f:Flight)
WITH f.flight_number + '/' + f.fleet_type_description AS flight_id,
     avg(j.arrival_delay_minutes) AS avg_arrival_delay
RETURN flight_id, avg_arrival_delay
ORDER BY avg_arrival_delay ASC
LIMIT 10;

//5. Query 5 - Average flown miles grouped by loyalty program level
MATCH (p:Passenger)-[:TOOK]->(j:Journey)
WITH p.loyalty_program_level AS loyalty_level,
     avg(j.actual_flown_miles) AS avg_actual_flown_miles
RETURN loyalty_level, avg_actual_flown_miles
ORDER BY avg_actual_flown_miles DESC;

//6. Query 6 - Passengers with overall satisfaction score greater than 3
MATCH (p:Passenger)-[:TOOK]->(j:Journey)
WITH p,
     j.food_satisfaction_score AS food,
     round(abs(j.arrival_delay_minutes / 20.0), 1) AS delay_ratio,
     round(j.number_of_legs * 1.5, 1) AS legs_factor,
     round(j.actual_flown_miles / 3000.0, 1) AS miles_ratio
WITH p,
     food,
     CASE
         WHEN delay_ratio < 0 THEN 5 - 0
         WHEN delay_ratio > 5 THEN 5 - 5
         ELSE 5 - delay_ratio
     END AS delay_score,
     CASE
         WHEN legs_factor < 0 THEN 5 - 0
         WHEN legs_factor > 5 THEN 5 - 5
         ELSE 5 - legs_factor
     END AS legs_score,
     CASE
         WHEN miles_ratio < 0 THEN 5 - 0
         WHEN miles_ratio > 5 THEN 5 - 5
         ELSE 5 - miles_ratio
     END AS miles_score
WITH p,
     round(0.5 * food +
     0.35 * delay_score +
     0.1 * legs_score +
     0.05 * miles_score, 1) AS overall_satisfaction_score
WHERE overall_satisfaction_score > 3
RETURN count(DISTINCT p) AS satisfied_passenger_count;
